import React, { useEffect, useMemo, useRef, useState } from "react";
import mermaid from "mermaid";

/**
 * Mermaid 렌더러 (단일 다이어그램)
 */
function MermaidDiagram({ code, id }: { code: string; id: string }) {
  const ref = useRef<HTMLDivElement | null>(null);
  useEffect(() => {
    if (!ref.current) return;
    // 매 다이어그램 렌더 시 초기화 (보안/테마 명시)
    mermaid.initialize({ startOnLoad: false, theme: "default", securityLevel: "strict" });
    const render = async () => {
      try {
        const { svg } = await mermaid.render(`m-${id}-${Date.now()}`, code);
        if (ref.current) ref.current.innerHTML = svg;
      } catch (e) {
        if (ref.current) ref.current.innerHTML = `<pre class='text-xs whitespace-pre-wrap'>${String(e)}</pre>`;
      }
    };
    render();
  }, [code, id]);
  return <div ref={ref} className="w-full overflow-auto rounded-xl bg-white p-4 shadow" />;
}

/** 공용 버튼 */
function Btn({ onClick, children, title }: { onClick?: () => void; children: React.ReactNode; title?: string }) {
  return (
    <button title={title} onClick={onClick} className="inline-flex items-center gap-2 rounded-2xl px-3 py-2 text-sm shadow-sm hover:shadow transition active:scale-[0.99] border border-slate-200">
      {children}
    </button>
  );
}

/** 유틸: 클립보드 복사 */
async function copy(text: string) {
  try { await navigator.clipboard.writeText(text); } catch {}
}

/** 유틸: 현재 컨테이너의 SVG 다운로드 */
function downloadSVGFromContainer(container: HTMLDivElement | null, filename = "diagram.svg") {
  if (!container) return;
  const svg = container.querySelector("svg");
  if (!svg) return;
  const blob = new Blob([svg.outerHTML], { type: "image/svg+xml;charset=utf-8" });
  const url = URL.createObjectURL(blob);
  const a = document.createElement("a");
  a.href = url; a.download = filename; a.click();
  URL.revokeObjectURL(url);
}
